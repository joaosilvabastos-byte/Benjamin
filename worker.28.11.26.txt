// Cloudflare Worker — proxy Benjamin -> Gemini, com timeout, fallback de modelos e JSON seguro
const CORS = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Methods': 'POST, OPTIONS',
  'Access-Control-Allow-Headers': 'Content-Type, X-Benjamin-Key',
  'Access-Control-Max-Age': '86400',   // o browser guarda a pré-verificação (OPTIONS) e não a repete em cada pergunta
};

// Ordem de tentativa. Série 2.5 já não está disponível para projetos novos (HTTP 404); a série 3.x usa thinkingLevel (não thinkingBudget).
// "minimal" = pensar o mínimo possível (a 3.x não tem "desligado"): respostas rápidas para conversa curta.
// Lista oficial de modelos/avisos: https://ai.google.dev/gemini-api/docs/deprecations
const MODELOS = [
  { nome: 'gemini-3.5-flash-lite', thinking: { thinkingLevel: 'minimal' } },
  { nome: 'gemini-3.6-flash',      thinking: { thinkingLevel: 'minimal' } },
];

const TIMEOUT_MS = 4000; // por tentativa
const HEDGE_MS = 2000;   // se o 1.º modelo ainda não respondeu a este tempo, arranca o 2.º em paralelo (ganha o que responder primeiro)

const SISTEMA =
  'Tu és o Benjamin, co-piloto de cabine para motoristas de pesados: direto, útil, com bom humor, ' +
  'em português de Portugal. Responde em 1 ou 2 frases curtas, naturais para serem ditas em voz alta, ' +
  'sem listas, sem markdown e sem emojis.';

function json(obj, status = 200) {
  return new Response(JSON.stringify(obj), {
    status,
    headers: { 'Content-Type': 'application/json', ...CORS },
  });
}

async function chamarModelo(modelo, texto, apiKey, externo, semThinking = false) {
  const ctrl = new AbortController();
  const timer = setTimeout(() => ctrl.abort(), TIMEOUT_MS);
  if (externo) externo.addEventListener('abort', () => ctrl.abort(), { once: true });

  const generationConfig = { maxOutputTokens: 400, temperature: 0.8 };   // só é um teto (a resposta termina sozinha em 1-2 frases); na série 3.x os tokens de raciocínio contam para o teto, e um teto pequeno devolve resposta vazia
  if (modelo.thinking && !semThinking) generationConfig.thinkingConfig = modelo.thinking;

  try {
    const res = await fetch(
      `https://generativelanguage.googleapis.com/v1beta/models/${modelo.nome}:generateContent`,
      {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'x-goog-api-key': apiKey },
        body: JSON.stringify({
          systemInstruction: { parts: [{ text: SISTEMA }] },
          contents: [{ role: 'user', parts: [{ text: texto }] }],
          generationConfig,
          // piadas/quiz: filtros só no nível alto (os valores por defeito podem bloquear humor inofensivo e devolver resposta vazia)
          safetySettings: ['HARM_CATEGORY_HARASSMENT', 'HARM_CATEGORY_HATE_SPEECH', 'HARM_CATEGORY_SEXUALLY_EXPLICIT', 'HARM_CATEGORY_DANGEROUS_CONTENT']
            .map((category) => ({ category, threshold: 'BLOCK_ONLY_HIGH' })),
        }),
        signal: ctrl.signal,
      }
    );

    // Lê como texto: o upstream pode devolver não-JSON (ex.: "error code: 524")
    const corpo = await res.text();
    let data = null;
    try { data = JSON.parse(corpo); } catch { /* não é JSON */ }

    // Se a API recusar o campo de raciocínio (400 a falar em "thinking"), repete uma vez sem ele em vez de falhar
    if (res.status === 400 && !semThinking && modelo.thinking && /thinking/i.test(corpo)) {
      return chamarModelo(modelo, texto, apiKey, externo, true);
    }
    if (!res.ok || !data) {
      throw new Error(`HTTP ${res.status}: ${(data?.error?.message || corpo).slice(0, 200)}`);
    }

    const cand = data.candidates?.[0];
    const resposta = cand?.content?.parts?.map((p) => p.text || '').join('').trim();
    if (!resposta) throw new Error(`Resposta vazia (finishReason=${cand?.finishReason || '-'}, bloqueio=${data.promptFeedback?.blockReason || '-'})`);
    return resposta;
  } finally {
    clearTimeout(timer);
  }
}


// ---------------------------------------------------------------------------
// SOS: e-mail ao encarregado com a localização (Resend, plano gratuito).
// Segredos (wrangler secret put ...): RESEND_API_KEY, SOS_KEY (a mesma que pões na app)
// Variável opcional: SOS_FROM  ex.: "Benjamin SOS <sos@teudominio.pt>"  (domínio verificado no Resend)
// ---------------------------------------------------------------------------
// REMETENTE DO E-MAIL SOS: troca aqui (ou define a variável SOS_FROM no Cloudflare) quando tiveres um domínio verificado no Resend.
// Ex.: 'Benjamin SOS <sos@teudominio.pt>'. Enquanto não, mantém-se o endereço de teste do Resend.
const SOS_FROM_PADRAO = 'Benjamin SOS <onboarding@resend.dev>';

const EMAIL_RE = /^[^\s@<>"]{1,64}@[^\s@<>"]{1,255}$/;

function esc(t) {
  return String(t).replace(/[&<>"]/g, (c) => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]));
}

async function tratarSos(request, env) {
  if (!env.SOS_KEY || !env.RESEND_API_KEY) return json({ ok: false, erro: 'sos-nao-configurado' }, 503);
  if (request.headers.get('X-Benjamin-Key') !== env.SOS_KEY) return json({ ok: false, erro: 'nao-autorizado' }, 401);

  let d;
  try { d = await request.json(); } catch { return json({ ok: false, erro: 'json' }, 400); }
  const para = String(d.para || '').trim();
  if (!EMAIL_RE.test(para)) return json({ ok: false, erro: 'email' }, 400);

  const condutor = String(d.condutor || 'Condutor').slice(0, 80);
  const quando = new Date(Number(d.ts) || Date.now()).toLocaleString('pt-PT', { timeZone: 'Europe/Lisbon' });
  const temPos = Number.isFinite(d.lat) && Number.isFinite(d.lon);
  const mapa = temPos ? `https://www.google.com/maps?q=${d.lat},${d.lon}` : null;
  const prec = Number.isFinite(d.prec) && d.prec > 0 ? ` (precisão ~${Math.round(d.prec)} m)` : '';

  const texto =
    `ALERTA SOS - ${condutor}\n` +
    `Hora: ${quando}\n` +
    (temPos ? `Localização: ${d.lat.toFixed(5)}, ${d.lon.toFixed(5)}${prec}\nMapa: ${mapa}\n`
            : `Localização: indisponível no momento do alerta.\n`) +
    `Origem: ${d.origem === 'voz' ? 'comando de voz' : 'botão SOS'}\n\n` +
    `Enviado automaticamente pelo Benjamin AI Co-pilot.`;
  const html = `<p><b>ALERTA SOS - ${esc(condutor)}</b></p><p>Hora: ${esc(quando)}</p>` +
    (temPos ? `<p>Localização: ${d.lat.toFixed(5)}, ${d.lon.toFixed(5)}${esc(prec)}<br><a href="${mapa}">Abrir no mapa</a></p>`
            : '<p>Localização: indisponível no momento do alerta.</p>') +
    `<p>Origem: ${d.origem === 'voz' ? 'comando de voz' : 'botão SOS'}</p><p><i>Enviado automaticamente pelo Benjamin AI Co-pilot.</i></p>`;

  const res = await fetch('https://api.resend.com/emails', {
    method: 'POST',
    headers: { Authorization: `Bearer ${env.RESEND_API_KEY}`, 'Content-Type': 'application/json' },
    body: JSON.stringify({
      from: env.SOS_FROM || SOS_FROM_PADRAO,
      to: [para],
      subject: `SOS - ${condutor} precisa de ajuda`,
      text: texto,
      html,
    }),
  });
  if (!res.ok) {
    console.error('Resend falhou:', res.status, (await res.text()).slice(0, 200));
    return json({ ok: false, erro: 'envio' }, 502);
  }
  return json({ ok: true });
}

export default {
  async fetch(request, env) {
    if (request.method === 'OPTIONS') return new Response(null, { headers: CORS });
    if (request.method !== 'POST') return json({ erro: true, resposta: null }, 405);

    if (new URL(request.url).pathname === '/sos') return tratarSos(request, env);

    let texto;
    try {
      ({ textoUsuario: texto } = await request.json());
    } catch {
      return json({ erro: true, resposta: null }, 400);
    }
    if (typeof texto !== 'string' || !texto.trim()) return json({ erro: true, resposta: null }, 400);
    texto = texto.trim().slice(0, 1200);   // os pedidos de piada/quiz levam instruções longas: com 500 eram cortadas a meio

    // 1.º modelo logo; o 2.º arranca quando o 1.º falha OU passados HEDGE_MS sem resposta (sem esperar o timeout inteiro)
    const parar = new AbortController();
    const erros = [];
    let feito = false;
    const tentar = async (modelo, esperar) => {
      if (esperar) await esperar();
      if (feito) throw new Error('já respondido');
      try {
        const resposta = await chamarModelo(modelo, texto, env.GEMINI_API_KEY, parar.signal);
        return { resposta, modelo: modelo.nome };
      } catch (e) {
        console.error(`[${modelo.nome}] falhou:`, e.message); // visível em wrangler tail / dashboard
        erros.push(`${modelo.nome}: ${String(e.message).slice(0, 160)}`);
        throw e;
      }
    };
    const p0 = tentar(MODELOS[0]);
    const p1 = tentar(MODELOS[1], () => Promise.race([new Promise((r) => setTimeout(r, HEDGE_MS)), p0.then(() => {}, () => {})]));
    try {
      const r = await Promise.any([p0, p1]);
      feito = true;
      parar.abort();
      return json({ erro: false, resposta: r.resposta, modelo: r.modelo });
    } catch { /* ambos falharam */ }

    // Nunca devolvemos mensagens técnicas como "resposta" (o Benjamin lê-as em voz alta)
    return json({ erro: true, resposta: null, detalhe: erros.join(' | ') }, 502);   // 'detalhe' só vai para o log da app, nunca é dito em voz alta
  },
};
